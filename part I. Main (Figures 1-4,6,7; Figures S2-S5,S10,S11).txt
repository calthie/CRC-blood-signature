clean_lod_symbols <- function(x) {
                                  if (is.numeric(x)) return(x) 
                                  x_str <- trimws(as.character(x)) 
                                  x_no_na <- x_str[!is.na(x_str)]
                                  if (length(x_no_na) == 0) return(as.numeric(x_str))                                
                                  is_valid <- grepl("^([<>]?)[-+]?[0-9]*\\.?[0-9]+$", x_no_na)
                                  if (all(is_valid)) {                                               
                                                     is_less <- grepl("^<", x_str)
                                                     num_val <- suppressWarnings(as.numeric(gsub("[<>]", "", x_str)))
                                                     num_val[is_less & !is.na(is_less)] <- num_val[is_less & !is.na(is_less)] / 2   
                                                     return(num_val)
                                                     } 
                                  else {                                                  
                                       return(x)
                                       }
                                 }
df_cleaned_symbols <- df %>% mutate(across(everything(), clean_lod_symbols))
df_numeric_fixed <- df_cleaned_symbols %>% mutate(across(where(is.integer), as.numeric))

get_mode <- function(v) { uniqv <- unique(na.omit(v)); uniqv[which.max(tabulate(match(v, uniqv)))] }
df_final <- df_filtered %>% mutate(

evaluate_strategy <- function(data_imp) {
                                         temp_data <- data_imp
                                         temp_data$Target <- as.factor(ifelse(grepl("4", temp_data[[target_col]]), 1, 0))
                                         cols_to_drop <- c("AJCC", "T", "N", "M","Tumor_site") 
                                         temp_data <- temp_data %>% dplyr::select(-any_of(cols_to_drop))
                                         set.seed(2026)
                                         rf_model <- randomForest(Target ~ ., data = temp_data, ntree = 500, importance = TRUE)
                                         oob_pred <- predict(rf_model, type = "prob")[, 2]
                                         roc_obj <- pROC::roc(temp_data$Target, oob_pred, quiet = TRUE)
                                         auc_val <- as.numeric(pROC::auc(roc_obj))  
                                         imp <- importance(rf_model)
                                         top_features <- rownames(imp)[order(imp[, "MeanDecreaseGini"], decreasing = TRUE)][1:10]
                                         return(list(AUC = auc_val, TopFeatures = top_features,ROC = roc_obj))
                                         }
res_A <- evaluate_strategy(df_strategy_A)

res_C <- evaluate_strategy(df_strategy_C)

get_roc_metrics <- function(roc_obj) {
                                      best_coords <- pROC::coords(roc_obj, "best",ret = c("accuracy", "sensitivity", "specificity"), best.method = "youden",transpose = FALSE)  
                                      acc <- best_coords$accuracy[1]
                                      sen <- best_coords$sensitivity[1]
                                      spe <- best_coords$specificity[1]
                                      auc_val <- pROC::auc(roc_obj)
                                      sprintf("AUC:%.3f (ACC:%.2f SEN:%.2f SPE:%.2f)", auc_val, acc, sen, spe)
                                      }
generate_full_metrics_table <- function(model_name, prob, actual_label) {
                                                                         roc_obj <- pROC::roc(actual_label, prob, quiet = TRUE)
                                                                         coords <- pROC::coords(roc_obj, "best", ret = c("threshold", "accuracy", "sensitivity", "specificity", "ppv", "npv"), 
                                                                                                best.method = "youden", transpose = FALSE)  
                                                                         auc_val <- as.numeric(pROC::auc(roc_obj))
                                                                         ci_auc  <- as.numeric(pROC::ci(roc_obj))
                                                                         pr_obj <- pr.curve(scores.class0 = prob[actual_label == 1],scores.class1 = prob[actual_label == 0])
                                                                         f1_score <- 2 * (coords$ppv[1] * coords$sensitivity[1]) / (coords$ppv[1] + coords$sensitivity[1])
                                                                         data.frame(
                                                                                    Model       = model_name,
                                                                                    ROC_AUC     = sprintf("%.3f (%.3f-%.3f)", auc_val, ci_auc[1], ci_auc[3]),
                                                                                    PR_AUC      = sprintf("%.3f", pr_obj$auc.integral),
                                                                                    Cutoff      = sprintf("%.3f", coords$threshold[1]),
                                                                                    Accuracy    = sprintf("%.3f", coords$accuracy[1]),
                                                                                    Sensitivity = sprintf("%.3f", coords$sensitivity[1]),
                                                                                    Specificity = sprintf("%.3f", coords$specificity[1]),
                                                                                    PPV         = sprintf("%.3f", coords$ppv[1]),
                                                                                    NPV         = sprintf("%.3f", coords$npv[1]),
                                                                                    F1_Score    = sprintf("%.3f", f1_score)
                                                                                    )
                                                                          }

calculate_subgroup_auc <- function(data, group_var, nice_name = group_var) {
                                                                            groups <- unique(data[[group_var]])
                                                                            groups <- groups[!is.na(groups) & groups != ""]
                                                                            res <- data.frame()
                                                                            for(g in groups) {
                                                                                              sub_data <- data[data[[group_var]] == g, ]
                                                                                              tryCatch({   
                                                                                                        if(length(unique(sub_data$Is_Stage_IV)) == 2 && nrow(sub_data) > 5) {
                                                                                                           roc_obj <- roc(sub_data$Is_Stage_IV, sub_data$Stage_IV, quiet = TRUE)
                                                                                                           ci_obj <- ci(roc_obj)
                                                                                                           res <- rbind(res, data.frame(
                                                                                                                                        Subgroup = paste(nice_name, g, sep = ": "), 
                                                                                                                                        N = nrow(sub_data), 
                                                                                                                                        AUC = as.numeric(ci_obj[2]), Lower = as.numeric(ci_obj[1]), Upper = as.numeric(ci_obj[3])
                                                                                                                                         ))
                                                                                                          }}, error = function(e){}) 
                                                                                               }
                                                                            return(res)
                                                                            }

calc_metrics_with_ci <- function(roc_obj, actual_labels, pred_probs, model_name) {
                                                                                  y_true <- as.numeric(as.character(actual_labels))  
                                                                                  auc_ci <- pROC::ci.auc(roc_obj)
                                                                                  auc_val <- auc_ci[2]; auc_low <- auc_ci[1]; auc_up <- auc_ci[3] 
                                                                                  best_coords <- pROC::coords(roc_obj, "best", ret = c("threshold"), transpose = FALSE)
                                                                                  thresh <- best_coords$threshold[1]
                                                                                  y_pred <- ifelse(pred_probs >= thresh, 1, 0)
                                                                                  TP <- sum(y_pred == 1 & y_true == 1)
                                                                                  TN <- sum(y_pred == 0 & y_true == 0)
                                                                                  FP <- sum(y_pred == 1 & y_true == 0)
                                                                                  FN <- sum(y_pred == 0 & y_true == 1)
                                                                                  acc_test  <- binom.test(TP + TN, TP + TN + FP + FN)
                                                                                  sens_test <- binom.test(TP, TP + FN)
                                                                                  spec_test <- binom.test(TN, TN + FP)
  
                       res <- data.frame(
                                         Model = model_name,
                                         AUC = sprintf("%.3f (%.3f-%.3f)", auc_val, auc_low, auc_up),
                                         Accuracy = sprintf("%.3f (%.3f-%.3f)", acc_test$estimate, acc_test$conf.int[1], acc_test$conf.int[2]),
                                         Sensitivity = sprintf("%.3f (%.3f-%.3f)", sens_test$estimate, sens_test$conf.int[1], sens_test$conf.int[2]),
                                         Specificity = sprintf("%.3f (%.3f-%.3f)", spec_test$estimate, spec_test$conf.int[1], spec_test$conf.int[2]),
                                         stringsAsFactors = FALSE
                                         )
                       return(res)
}

clean_lod_symbols <- function(x) {
                                  if (is.numeric(x)) return(x) 
                                  x_str <- trimws(as.character(x)) 
                                  x_no_na <- x_str[!is.na(x_str)]
                                  if (length(x_no_na) == 0) return(as.numeric(x_str))                                
                                  is_valid <- grepl("^([<>]?)[-+]?[0-9]*\\.?[0-9]+$", x_no_na)
                                  if (all(is_valid)) {                                               
                                                      is_less <- grepl("^<", x_str)
                                                      num_val <- suppressWarnings(as.numeric(gsub("[<>]", "", x_str)))
                                                      num_val[is_less & !is.na(is_less)] <- num_val[is_less & !is.na(is_less)] / 2   
                                                      return(num_val)
                                                     } 
                                  else {                                                  
                                        return(x)
                                       }
                                 }

## DCA 
generate_dca_plot <- function(df, cutoff, cohort_label, out_path) {
                                                                   thresholds <- seq(0, 0.40, by = 0.01) 
                                                                   N <- nrow(df)
                                                                   prev <- sum(df$OP_3y, na.rm = TRUE) / N  
                                                                   dca_data <- data.frame(threshold = thresholds)
                                                                   dca_data$Treat_All <- prev - (1 - prev) * (thresholds / (1 - thresholds))
                                                                   dca_data$Treat_None <- 0
                                                                   dca_data$Model <- sapply(thresholds, function(pt) {
                                                                                                                      tp <- sum(df$Nomogram_Prob >= pt & df$OP_3y == 1, na.rm = TRUE)
                                                                                                                      fp <- sum(df$Nomogram_Prob >= pt & df$OP_3y == 0, na.rm = TRUE)
                                                                                                                      (tp / N) - (fp / N) * (pt / (1 - pt))
                                                                                                                      }}
  
dca_long <- dca_data %>% pivot_longer(cols = c("Treat_All", "Treat_None", "Model"), names_to = "Strategy", values_to = "Net_Benefit") %>%
            mutate(Strategy = factor(Strategy, levels = c("Treat_None", "Treat_All", "Model"), labels = c("Treat None", "Treat All", "Nomogram")))
nb_at_cutoff <- dca_data$Model[which.min(abs(dca_data$threshold - cutoff))]
p_dca <- ggplot(dca_long, aes(x = threshold, y = Net_Benefit, color = Strategy)) +
         geom_line(linewidth = 1) +
         scale_color_manual(values = c("Treat None" = "black", "Treat All" = "grey50", "Nomogram" = "#E41A1C")) +
         coord_cartesian(ylim = c(-0.02, max(dca_data$Model, na.rm = TRUE) * 1.2), xlim = c(0, 0.4)) +
         geom_vline(xintercept = cutoff, linetype = "dashed", color = "#377EB8", linewidth = 0.8) +
         geom_point(data = data.frame(x = cutoff, y = nb_at_cutoff), aes(x = x, y = y), color = "#377EB8", size = 3, inherit.aes = FALSE) +
         annotate("text", x = cutoff + 0.01, y = max(dca_data$Model, na.rm=T) * 0.9, 
                   label = sprintf("Optimal Cutoff: %.2f%%\nMax Net Benefit", cutoff * 100), 
                   color = "#377EB8", hjust = 0, size = 4, fontface = "bold") +
         theme_classic(base_size = 13) +
         labs(title = paste0(cohort_label, ": Decision Curve"), x = "Threshold Probability", y = "Net Benefit") +
         theme(legend.position = c(0.8, 0.8), legend.title = element_blank(), legend.background = element_rect(fill="transparent"))
         ggsave(out_path, plot = p_dca, width = 5.5, height = 5.5)
         return(p_dca)
}

plot_time_roc <- function(d, cohort_label, file_path, cutoff) {
                                                               clean_d <- d %>% filter(!is.na(OP_3y), !is.na(Nomogram_Prob))
                                                               r <- roc(clean_d$OP_3y, clean_d$Nomogram_Prob, quiet = TRUE)
                                                               ci_res <- as.numeric(ci.auc(r))
                                                               coords_res <- pROC::coords(r, x = cutoff, input = "threshold", ret = c("sensitivity", "specificity"), transpose = FALSE)
                                                               sens <- coords_res$sensitivity[1]
                                                               spec <- coords_res$specificity[1]
                                                               wt <- wilcox.test(Nomogram_Prob ~ OP_3y, data = clean_d)
                                                               pval <- wt$p.value
                                                               p_str <- ifelse(pval < 0.001, "P < 0.001", sprintf("P = %.4f", pval))
                                                               anno_text <- sprintf("AUC = %.3f (%.3f-%.3f)\n%s\nSens = %.1f%%\nSpec = %.1f%%",ci_res[2], ci_res[1], ci_res[3], p_str, sens * 100, spec * 100)
                                                               roc_list <- list("3-Year" = r)
                                                               p_roc <- ggroc(roc_list, linewidth = 1) + 
                                                                        scale_color_manual(values = c("#E41A1C")) + 
                                                                        geom_abline(slope = 1, intercept = 1, linetype = "dashed", color = "grey50") +
                                                                        annotate("text", x = 0.45, y = 0.25, label = anno_text, hjust = 0, size = 4.5, fontface = "bold") + 
                                                                        theme_classic(base_size = 13) + 
                                                                        theme(legend.position = "none") +
                                                                        labs(title = paste0(cohort_label, ": ROC Curve"), x = "Specificity", y = "Sensitivity")
                                                               ggsave(file_path, plot = p_roc, width = 5.5, height = 5.5)
                                                               data.frame(
                                                                          Cohort = cohort_label, Timepoint = "3-Year", N = nrow(clean_d), Events = sum(clean_d$OP_3y == 1),
                                                                          AUC_95CI = sprintf("%.3f (%.3f-%.3f)", ci_res[2], ci_res[1], ci_res[3]),
                                                                          Sensitivity = sprintf("%.1f%%", sens * 100),
                                                                          Specificity = sprintf("%.1f%%", spec * 100),
                                                                          P_value = signif(pval, 4)
                                                                          )
                                                               }

auc_all <- bind_rows(
                     plot_time_roc(internal_data, "Internal Cohort", file.path(out_dir, "Internal_Recurrence_ROC.pdf"), optimal_cutoff),
                     plot_time_roc(external_data, "External Cohort", file.path(out_dir, "External_Recurrence_ROC.pdf"), optimal_cutoff)
                    )
