# LRT AND Model Fit Comparison
data_lrt <- data_surv_merged[, c("Sample_ID", "Risk_Score", "Age", "Actual_Stage", "CEA", "Time_month", "Overall_survival")]
data_lrt <- na.omit(data_lrt)
data_lrt$OS_time <- as.numeric(data_lrt$Time_month)
data_lrt$OS_status <- as.numeric(data_lrt$Overall_survival)
data_lrt$Age <- factor(data_lrt$Age, levels = c("< 65", ">= 65"))
data_lrt$Actual_Stage <- factor(data_lrt$Actual_Stage, levels = c("Stage_I_II", "Stage_III"))

data_lrt$log_CEA <- log2(data_lrt$CEA + 1)
model_base <- coxph(Surv(OS_time, OS_status) ~ Actual_Stage + Age, data = data_lrt)
model_cea  <- coxph(Surv(OS_time, OS_status) ~ Actual_Stage + Age + log_CEA, data = data_lrt)
model_risk <- coxph(Surv(OS_time, OS_status) ~ Actual_Stage + Age + Risk_Score, data = data_lrt)
model_both <- coxph(Surv(OS_time, OS_status) ~ Actual_Stage + Age + log_CEA + Risk_Score, data = data_lrt)

cat("\n--- 1. AIC Comparison (Lower is better) ---\n")
aic_results <- AIC(model_base, model_cea, model_risk)
print(aic_results)

cat("\n--- 2. Likelihood Ratio Test (LRT) ---\n")
cat("LRT: Base Model vs. Model with CEA\n")
print(anova(model_base, model_cea))

cat("\nLRT: Base Model vs. Model with Risk Score\n")
print(anova(model_base, model_risk))

cat("\nLRT: Model with CEA vs. Combined Model (CEA + Risk Score)\n")
lrt_incremental <- anova(model_cea, model_both)
print(lrt_incremental)

p_val_incremental <- lrt_incremental[2, ncol(lrt_incremental)]
