library(Seurat)
library(ggplot2)
library(DoubletFinder) 
Run_Strict_QC_Pipeline <- function(seu_obj, dataset_name) {  
                                                          seu_obj[["percent.mt"]] <- PercentageFeatureSet(seu_obj, pattern = "^MT-")
                                                          seu_obj[["percent.hb"]] <- PercentageFeatureSet(seu_obj, pattern = "^HB[^(P)]") 
                                                          seu_obj[["percent.ribo"]] <- PercentageFeatureSet(seu_obj, pattern = "^RP[SL]")
p_qc <- VlnPlot(seu_obj, features = c("nFeature_RNA", "nCount_RNA", "percent.mt", "percent.ribo", "percent.hb"), ncol = 5, pt.size = 0, group.by = "GSM_ID") 
p_qc_modified <- p_qc & scale_y_continuous(breaks = scales::pretty_breaks(n = 12)) & theme(panel.grid.major.y = element_line(color = "grey80", linetype = "dashed")) 
print(p_qc_modified) 
  
pdf_name <- paste0("QC_VlnPlot_BeforeFilter_", dataset_name, ".pdf")
ggsave(pdf_name, plot = p_qc_modified, width = 18, height = 5)
seu_filtered <- subset(seu_obj, subset = nFeature_RNA > 200 & nFeature_RNA < 6000 & percent.mt < 25 & percent.hb < 1)
sce_list <- SplitObject(seu_filtered, split.by = "GSM_ID")
for (i in 1:length(sce_list)) {
                              obj <- sce_list[[i]]
                              obj <- NormalizeData(obj, verbose = FALSE)
                              obj <- FindVariableFeatures(obj, verbose = FALSE)
                              obj <- ScaleData(obj, verbose = FALSE)
                              obj <- RunPCA(obj, verbose = FALSE)
                              obj <- RunUMAP(obj, dims = 1:15, verbose = FALSE)
                              obj <- FindNeighbors(obj, dims = 1:15, verbose = FALSE)
                              obj <- FindClusters(obj, resolution = 0.5, verbose = FALSE)
    
                              sweep.res <- paramSweep(obj, PCs = 1:15, sct = FALSE)
                              sweep.stats <- summarizeSweep(sweep.res, GT = FALSE)
                              bcmvn <- find.pK(sweep.stats)
                              optimal_pK <- as.numeric(as.character(bcmvn$pK[which.max(bcmvn$BCmetric)]))

                              nExp_poi <- round(ncol(obj) * 8e-6 * ncol(obj)) 
                              homotypic.prop <- modelHomotypic(obj@meta.data$seurat_clusters)
                              nExp_poi.adj <- round(nExp_poi * (1 - homotypic.prop))
    
                              obj <- doubletFinder(obj, PCs = 1:15, pN = 0.25, pK = optimal_pK, nExp = nExp_poi.adj, sct = FALSE)
                              df_col <- colnames(obj@meta.data)[grepl("DF.classifications", colnames(obj@meta.data))]
                              obj$Doublet_Singlet <- obj@meta.data[[df_col]]
                              obj@meta.data <- obj@meta.data[, !grepl("^DF.classifications|^pANN", colnames(obj@meta.data))]
                              sce_list[[i]] <- obj
                              } 
seu_clean <- merge(sce_list[[1]], y = sce_list[-1])
print(table(seu_clean$Doublet_Singlet))
seu_clean <- subset(seu_clean, subset = Doublet_Singlet == "Singlet")  
seu_clean$Doublet_Singlet <- NULL 
gc()
return(seu_clean)
}

seu_178318_clean <- Run_Strict_QC_Pipeline(seu_obj = seu_178318_clean, dataset_name = "GSE178318")
seu_245552_clean <- Run_Strict_QC_Pipeline(seu_obj = seu_245552_clean, dataset_name = "GSE245552")

get_sample_details <- function(seu_obj, dataset_name) {
                                                       meta <- seu_obj@meta.data
                                                       sample_ids <- unique(meta$GSM_ID)
                                                       result_df <- data.frame()
for (id in sample_ids) {
                       sub_meta <- meta[meta$GSM_ID == id, ]
                       tmp_df <- data.frame(
                                            Dataset = dataset_name,
                                            GSM_ID = id,
                                            TNM_Stage = unique(sub_meta$TNM_Stage)[1],     
                                            Stage_AJCC = unique(sub_meta$Stage_AJCC)[1],
                                            Cell_Count = nrow(sub_meta),                  
                                            Median_Genes = median(sub_meta$nFeature_RNA),  
                                            Median_UMIs = median(sub_meta$nCount_RNA)    
                                            )
                       result_df <- rbind(result_df, tmp_df)
                       }  
return(result_df)
}

analysis_grid <- data.frame(
                            CellType = c(rep("Epithelial_cells", 3), "Fibroblasts"),
                            Pathway = c("C5DC.Production", "BUN.Generation", "CA19.9.Synthesis", "C5DC.Production"),
                            stringsAsFactors = FALSE
                            )

target_cells <- unique(analysis_grid$CellType)
target_metrics <- unique(analysis_grid$Pathway)
df_meta_sc <- CRC_Mega@meta.data %>% filter(CellType %in% target_cells & Stage_AJCC %in% c("I", "II", "III", "IV")) %>% mutate(Stage_Group = ifelse(Stage_AJCC %in% c("I", "II", "III"), "I-III", "IV"),
              Stage_Group = factor(Stage_Group, levels = c("I-III", "IV"))) %>% dplyr::select(CellType, Stage_Group, Dataset, all_of(target_metrics))

safe_scale <- function(x) {
                           if (sd(x, na.rm = TRUE) == 0) return(rep(0, length(x)))
                           return(scale(x)[, 1])
                           }
df_norm <- df_meta_sc %>% group_by(Dataset, CellType) %>% mutate(across(all_of(target_metrics), ~ safe_scale(.))) %>% ungroup()
stats_list_wilcox <- list()
for (i in 1:nrow(analysis_grid)) {
                                  current_cell <- analysis_grid$CellType[i]
                                  current_pathway <- analysis_grid$Pathway[i]
                                  df_sub <- df_norm %>% filter(CellType == current_cell)
                                  val_I_III <- df_sub[[current_pathway]][df_sub$Stage_Group == "I-III"]
                                  val_IV    <- df_sub[[current_pathway]][df_sub$Stage_Group == "IV"]
                                  val_I_III <- val_I_III[!is.na(val_I_III)]
                                  val_IV    <- val_IV[!is.na(val_IV)]
                                  if(length(val_I_III) < 3 | length(val_IV) < 3) next
                                  p_val <- wilcox.test(val_I_III, val_IV)$p.value
                                  mean_diff <- mean(val_IV) - mean(val_I_III)
                                  se_diff <- sqrt(var(val_IV)/length(val_IV) + var(val_I_III)/length(val_I_III))
  
                                  stats_list_wilcox[[i]] <- data.frame(
                                                                       CellType = current_cell,
                                                                       Pathway = current_pathway,
                                                                       Estimate = mean_diff,
                                                                       Std_Error = se_diff,
                                                                       p_val = p_val
                                                                       )
                                  }

                           if (sd(x, na.rm=TRUE) == 0) return(rep(0, length(x))) else return(scale(x)[,1])
                          }
